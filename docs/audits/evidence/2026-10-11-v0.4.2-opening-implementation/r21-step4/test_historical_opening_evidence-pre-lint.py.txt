def api():
    assert hasattr(chstockdata, 'fetch_historical_opening_evidence'), 'public R2.1 capability missing'
    from chstockdata import opening_evidence
    return opening_evidence


def records(code, day):
    found = []
    for name, sha in SHAS.items():
        raw = (TDX / name).read_bytes()
        assert hashlib.sha256(raw).hexdigest() == sha
        found.extend(r for r in map(json.loads, raw.decode('utf-8').splitlines())
                     if r['security'] == code and r['date'] == day.replace('-', ''))
    return found


class ReplayClient:
    def __init__(self, rows):
        self.rows = rows
        self.client = self
    def get_history_minute_time_data(self, market, code, date):
        return copy.deepcopy(next(r['rows'] for r in self.rows if r['command'].endswith('fb4')))
    def get_history_transaction_data(self, market, code, start, count, date):
        row = next(r for r in self.rows if r['command'].endswith('fb5') and r['parameters']['start'] == start)
        assert (market, code, date, count) == (row['market_id'], row['security'], int(row['date']), 800)
        return copy.deepcopy(row['rows'])
    def close(self):
        pass


class Response:
    def __init__(self, payload, status=200):
        self.content = payload if isinstance(payload, bytes) else json.dumps(payload).encode()
        self.status_code = status
        self.headers = {'Content-Type': 'application/json'}
    def raise_for_status(self):
        if self.status_code != 200:
            raise ConnectionError('fault transport')
    def json(self):
        return json.loads(self.content)


class ReplayHttp:
    def __init__(self, payloads):
        self.payloads = list(payloads)
        self.requests = []
    def get(self, url, **kwargs):
        self.requests.append((url, kwargs))
        payload = self.payloads.pop(0)
        if isinstance(payload, Exception):
            raise payload
        return Response(payload)
    post = get


def empty(code):
    return {'announceCount': 0, 'data': []} if code.startswith(('0', '1')) else {
        'result': [], 'pageHelp': {'pageNo': 1, 'pageCount': 0, 'pageSize': 25, 'total': 0, 'data': []}}


def readers(code, day, official=None, mutate=None, versions=('0.11.7', '0.2.7')):
    m = api()
    rs = records(code, day)
    if mutate:
        mutate(rs)
    pages = [r for r in rs if r['command'].endswith('fb5')]
    # Replay a certified bounded page containing all opening rows, or an empty/late control.
    chosen = next((r for r in pages if any(x.get('time') == '09:30' for x in (r['rows'] or []))), pages[-1])
    client = ReplayClient(rs)
    return (m.TdxOpeningReader(client_factory=lambda **kw: client, versions=versions,
                               page_offsets=(chosen['parameters']['start'],)),
            m.OfficialOpeningReader(session=ReplayHttp(official or [empty(code)])))


def fetch(code='510300', day='2026-09-11', **kwargs):
    tdx, official = readers(code, day, kwargs.pop('official', None), kwargs.pop('mutate', None),
                            kwargs.pop('versions', ('0.11.7', '0.2.7')))
    return chstockdata.fetch_historical_opening_evidence(code, day, tdx_reader=tdx,
                                                        official_reader=official, **kwargs)

    assert data['target_execution_at'] == '2026-09-11T09:30:00+08:00'
    assert data['execution_effect_window']['end'] == '2026-09-11T09:31:00+08:00'

    assert datetime.fromisoformat(result.metadata.retrieved_at).utcoffset().total_seconds() == 0
    assert result.metadata.capability == 'historical_opening_evidence'
    assert {a.capability for a in result.metadata.attempts} >= {'mootdx:historical_opening_activity'}

@pytest.mark.parametrize('mode', ['auction_only','zero','minute_only','bad_price','bad_time','double_normalized'])
def test_no_qualified_activity_unknown(mode):
    def mutate(rs):
        for r in rs:
            if r['command'].endswith('fb5'):
                if mode=='auction_only': r['rows']=[x for x in r['rows'] if x['time']=='09:25']
                if mode=='minute_only': r['rows']=[]
                if mode=='zero':
                    for x in r['rows']: x['vol']=0
                if mode=='bad_price':
                    for x in r['rows']: x['price']=-1
                if mode=='bad_time':
                    for x in r['rows']: x['time']='09:30:01'
                if mode=='double_normalized':
                    for x in r['rows']: x['price']=x['price']/10
    assert fetch(mutate=mutate).data['opening_status'] == 'UNKNOWN'

def test_complete_current_pagination_and_partial_cap():

        official_reader=m.OfficialOpeningReader(session=http,page_size=1,max_pages=1))
